    // 3. Musik Kotak Musik Ulang Tahun Otomatis (Web Audio API)
    let audioCtx;
    function playBirthdayMelody() {
      try {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const notes = [
          [261.6, 0.3], [261.6, 0.3], [293.7, 0.5], [261.6, 0.5], [349.2, 0.5], [329.6, 0.9],
          [261.6, 0.3], [261.6, 0.3], [293.7, 0.5], [261.6, 0.5], [392.0, 0.5], [349.2, 0.9]
        ];
        let time = audioCtx.currentTime;
        notes.forEach(([freq, dur]) => {
          const osc = audioCtx.createOscillator();
          const gain = audioCtx.createGain();
          osc.type = 'sine';
          osc.frequency.value = freq;
          gain.gain.setValueAtTime(0.12, time);
          gain.gain.exponentialRampToValueAtTime(0.001, time + dur);
          osc.connect(gain);
          gain.connect(audioCtx.destination);
          osc.start(time);
          osc.stop(time + dur);
          time += dur + 0.06;
        });
      } catch (e) {}
    }

    async function enableMicBlow() {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        const ctx = new AudioContext();
        const src = ctx.createMediaStreamSource(stream);
        const analyser = ctx.createAnalyser();
        src.connect(analyser);
        const data = new Uint8Array(analyser.frequencyBinCount);
        document.getElementById('micBtn').innerText = '??? Sekarang Tiup ke arah HP kamu...';

        function checkBlow() {
          analyser.getByteFrequencyData(data);
          let sum = 0;
          for (let i = 0; i < data.length; i++) sum += data[i];
          if (sum / data.length > 42) {
            blowCandles();
            stream.getTracks().forEach(t => t.stop());
          } else {
            requestAnimationFrame(checkBlow);
          }
        }
        checkBlow();
      } catch (err) {
        alert('Mikrofon belum diizinkan, langsung sentuh api lilinnya saja ya Bebebcu! ??');
      }
    }
